#include <Wire.h>
#include <math.h>
#include <Preferences.h>

#define AD5933_ADDR 0x0D

const double CLK_FREQ = 16776000.0;
const double R_CAL = 979.0;
const double RFB = 975.0;

long startFreq = 10000;
long freqIncr = 250;
int numIncr = 360;

#define WINDOW_SIZE 5
#define NUM_POINTS 364
#define SAMPLES_PER_FREQ 1

struct Measurement {
  double imp, phase, real, imag, mag;
};

Measurement windows[NUM_POINTS][WINDOW_SIZE];
int windowIndices[NUM_POINTS];
int windowCounts[NUM_POINTS];

double gainFactorTable[NUM_POINTS];
double phaseOffsetTableDeg[NUM_POINTS];

Preferences preferences;

void resetI2C() {
  Wire.end();
  delay(100);
  Wire.begin(21, 22);
  Wire.setClock(400000);
  delay(200);
}

bool testAD5933() {
  Wire.beginTransmission(AD5933_ADDR);
  byte error = Wire.endTransmission();
  return error == 0;
}

void writeReg(byte reg, byte val) {
  Wire.beginTransmission(AD5933_ADDR);
  Wire.write(reg);
  Wire.write(val);
  Wire.endTransmission();
}

void writeReg24(byte reg, long val) {
  writeReg(reg,     (val >> 16) & 0xFF);
  writeReg(reg + 1, (val >> 8) & 0xFF);
  writeReg(reg + 2, val & 0xFF);
}

void setFrequency(byte reg, long freqHz) {
  long val = (long)((freqHz / (CLK_FREQ / 4.0)) * pow(2.0, 27.0));
  writeReg24(reg, val);
}

byte readStatus() {
  Wire.beginTransmission(AD5933_ADDR);
  Wire.write(0x8F);
  Wire.endTransmission();
  Wire.requestFrom(AD5933_ADDR, 1);
  return Wire.available() ? Wire.read() : 0;
}

int16_t readData(byte reg) {
  Wire.beginTransmission(AD5933_ADDR);
  Wire.write(reg);
  Wire.endTransmission();
  Wire.requestFrom(AD5933_ADDR, 2);
  if (Wire.available() < 2) return 0;
  return (int16_t)((Wire.read() << 8) | Wire.read());
}

bool waitValid(int max_ms = 3000) {
  unsigned long timeout = millis() + max_ms;
  while (millis() < timeout) {
    if (readStatus() & 0x02) return true;
    delay(5);
  }
  return false;
}

void calculateMovingAverage(int freqIndex, Measurement* avg) {
  double sumImp = 0, sumPhase = 0, sumReal = 0, sumImag = 0, sumMag = 0;
  int count = windowCounts[freqIndex];
  if (count <= 0) return;

  for (int i = 0; i < count; i++) {
    sumImp += windows[freqIndex][i].imp;
    sumPhase += windows[freqIndex][i].phase;
    sumReal += windows[freqIndex][i].real;
    sumImag += windows[freqIndex][i].imag;
    sumMag += windows[freqIndex][i].mag;
  }

  avg->imp = sumImp / count;
  avg->phase = sumPhase / count;
  avg->real = sumReal / count;
  avg->imag = sumImag / count;
  avg->mag = sumMag / count;
}

void initMovingAverageBuffers() {
  for (int i = 0; i < NUM_POINTS; i++) {
    windowIndices[i] = 0;
    windowCounts[i] = 0;
    gainFactorTable[i] = 0.0;
    phaseOffsetTableDeg[i] = 0.0;
    for (int j = 0; j < WINDOW_SIZE; j++) {
      windows[i][j] = {0, 0, 0, 0, 0};
    }
  }
}

void loadCalibration() {
  preferences.begin("ad5933", true);
  for (int i = 0; i < NUM_POINTS; i++) {
    char keyG[12];
    char keyP[12];
    snprintf(keyG, sizeof(keyG), "g%03d", i);
    snprintf(keyP, sizeof(keyP), "p%03d", i);
    gainFactorTable[i] = preferences.getDouble(keyG, 0.0);
    phaseOffsetTableDeg[i] = preferences.getDouble(keyP, 0.0);
  }
  preferences.end();
}

void saveCalibration() {
  preferences.begin("ad5933", false);
  for (int i = 0; i < NUM_POINTS; i++) {
    char keyG[12];
    char keyP[12];
    snprintf(keyG, sizeof(keyG), "g%03d", i);
    snprintf(keyP, sizeof(keyP), "p%03d", i);
    preferences.putDouble(keyG, gainFactorTable[i]);
    preferences.putDouble(keyP, phaseOffsetTableDeg[i]);
  }
  preferences.end();
}

bool hasValidCalibration() {
  for (int i = 0; i <= numIncr && i < NUM_POINTS; i++) {
    if (gainFactorTable[i] <= 0.0) return false;
  }
  return true;
}

void resetSweep() {
  writeReg(0x80, 0x10);
  delay(50);
  writeReg(0x80, 0xB0);
  delay(50);
  writeReg(0x80, 0x10);
  delay(50);
}

bool startSweep() {
  writeReg(0x80, 0xB0);
  delay(50);
  writeReg(0x80, 0x11);
  delay(100);
  writeReg(0x80, 0x21);
  delay(50);
  return true;
}

bool calibrateSweep() {
  resetSweep();
  startSweep();

  long currentFreq = startFreq;

  for (int i = 0; i <= numIncr && i < NUM_POINTS; i++) {
    if (!waitValid(5000)) return false;

    double realCal = (double)readData(0x94);
    double imagCal = (double)readData(0x96);
    double magCal = sqrt(realCal * realCal + imagCal * imagCal);

    if (magCal <= 0.0) return false;

    gainFactorTable[i] = (1.0 / R_CAL) / magCal;
    phaseOffsetTableDeg[i] = atan2(imagCal, realCal) * 180.0 / PI;

    Serial.printf("CAL,%ld,GF=%.12f,PH=%.12f\n",
                  currentFreq, gainFactorTable[i], phaseOffsetTableDeg[i]);

    if (i < numIncr) {
      writeReg(0x80, 0x31);
      currentFreq += freqIncr;
    } else {
      writeReg(0x80, 0xB1);
    }
  }

  return true;
}

void setup() {
  Serial.begin(115200);
  delay(2000);

  resetI2C();
  if (!testAD5933()) {
    while (1) delay(1000);
  }

  initMovingAverageBuffers();

  setFrequency(0x82, startFreq);
  setFrequency(0x85, freqIncr);
  writeReg(0x88, (numIncr >> 8) & 0xFF);
  writeReg(0x89, numIncr & 0xFF);

  loadCalibration();
  if (!hasValidCalibration()) {
    Serial.println("Calibrando sweep completo...");
    if (calibrateSweep()) {
      saveCalibration();
      Serial.println("Calibracao concluida.");
    } else {
      Serial.println("Falha na calibracao.");
    }
  }

  Serial.println("Freq(Hz),Imp_AVG(Ohm),Fase_AVG(°),Real_AVG(Ohm),Imag_AVG(Ohm),Mag_AVG");
}

void loop() {
  if (!hasValidCalibration()) {
    Serial.println("Sem calibracao valida. Recalibrando...");
    if (calibrateSweep()) {
      saveCalibration();
    }
    delay(500);
    return;
  }

  long currentFreq = startFreq;
  double somaImpMedia = 0.0;
  int pontosValidos = 0;

  resetSweep();
  startSweep();

  for (int i = 0; i <= numIncr && i < NUM_POINTS; i++) {
    if (!waitValid(2000)) break;

    double realSum = 0.0;
    double imagSum = 0.0;
    int validSamples = 0;

    for (int k = 0; k < SAMPLES_PER_FREQ; k++) {
      delay(2);

      double real = (double)readData(0x94);
      double imag = (double)readData(0x96);

      realSum += real;
      imagSum += imag;
      validSamples++;
    }

    if (validSamples > 0) {
      double realAvgRaw = realSum / validSamples;
      double imagAvgRaw = imagSum / validSamples;
      double magRaw = sqrt(realAvgRaw * realAvgRaw + imagAvgRaw * imagAvgRaw);

      if (magRaw > 0.0 && gainFactorTable[i] > 0.0) {
        double impedance = 1.0 / (magRaw * gainFactorTable[i]);
        double phaseRawDeg = atan2(imagAvgRaw, realAvgRaw) * 180.0 / PI;
        double phaseCorrDeg = phaseRawDeg - phaseOffsetTableDeg[i];
        double phaseCorrRad = phaseCorrDeg * PI / 180.0;

        double Z_real = impedance * cos(phaseCorrRad);
        double Z_imag = impedance * sin(phaseCorrRad);

        int idx = i;
        int pos = windowIndices[idx];

        windows[idx][pos] = {impedance, phaseCorrDeg, Z_real, Z_imag, magRaw};

        windowIndices[idx] = (windowIndices[idx] + 1) % WINDOW_SIZE;
        if (windowCounts[idx] < WINDOW_SIZE) windowCounts[idx]++;

        Measurement avg = {0, 0, 0, 0, 0};
        calculateMovingAverage(idx, &avg);

        Serial.printf("%ld,%.3f,%.12f,%.3f,%.3f,%.3f\n",
                      currentFreq, avg.imp, avg.phase, avg.real, avg.imag, avg.mag);

        somaImpMedia += avg.imp;
        pontosValidos++;
      }
    }

    if (i < numIncr) {
      writeReg(0x80, 0x31);
      currentFreq += freqIncr;
    } else {
      writeReg(0x80, 0xB1);
    }
  }

  if (pontosValidos > 0) {
    Serial.printf("MEDIA_IMPEDANCIA,%.3f Ohm,%d pontos\n",
                  somaImpMedia / pontosValidos, pontosValidos);
  }

  delay(100);
}
